cmake_minimum_required(VERSION 3.16)
project(luckfox_mavlink_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(mavlink_bridge src/main.cpp)
target_compile_options(mavlink_bridge PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS mavlink_bridge RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME bridge_integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_bridge.py $<TARGET_FILE:mavlink_bridge>)
  set_tests_properties(bridge_integration PROPERTIES TIMEOUT 45)
endif()
